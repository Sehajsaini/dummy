"""
=========================================================================================
 ML 102 - DRUG ADOPTION CAPSTONE  |  END-TO-END PRODUCTION PIPELINE  (FIXED v2)
=========================================================================================
Objective : Predict which of the 1,502 non-adopted physicians (Test_physicians.csv) will
            adopt drug 'XYZ' in Quarter 11, using 10 quarters of historical transactional
            data (Input_data_file1.csv) and static physician profile data
            (Input_data_file2.csv).

WHAT CHANGED FROM v1 (and why) -- read this first
-------------------------------------------------------------------------------------------
BUG 1 - SMOTE was run on top of fillna(-999).
  SMOTE synthesizes new minority-class rows by linearly interpolating between real
  neighbors. When a feature slot held the missing-value sentinel -999 next to a real
  value like 12, SMOTE invented meaningless synthetic values (e.g. -493) for that
  feature. The model was trained on thousands of these fabricated points -> muddy
  decision boundary -> poor recall and badly miscalibrated probabilities on truly
  unseen data (like the Q11 scoring population).
  FIX: SMOTE removed entirely. Class imbalance is now handled the way tree models are
  designed to handle it: `scale_pos_weight` (XGBoost) / LightGBM's equivalent, tuned by
  Optuna. No synthetic rows, no sentinel pollution.

BUG 2 - Missing values were hard-coded to -999 everywhere (including for SHAP).
  Both XGBoost and LightGBM handle NaN natively and correctly (they learn the best
  default split direction for missing values during training). Overwriting NaN with
  -999 forces the trees to treat "missing" as "a real number smaller than everything
  else", which is a false signal.
  FIX: NaN is now passed through untouched to both models and to SHAP. -999 fillna is
  used ONLY for VIF/correlation diagnostics, which need a placeholder and are not part
  of the actual model.

BUG 3 - Fixed 0.5 classification threshold.
  Adoption is a rare event. With a realistic (low) base rate, the vast majority of
  predicted probabilities sit well below 0.5 even for a well-performing model, so a
  flat 0.5 cutoff silently throws away almost every prediction (this is exactly why the
  Q11 scoring file only flagged 8 of 1,502 physicians as adopters).
  FIX: A proper decision threshold is now learned from out-of-fold rolling-CV
  predictions (never from the test set - that would leak), optimizing F-beta (beta is
  configurable; beta=1.5 default, meaning we weight recall higher than precision, but
  not so aggressively that precision collapses). This threshold is then applied
  consistently to train, test AND the Q11 scoring set. A full threshold-scan table is
  also printed so you can see the precision/recall/count trade-off at every cutoff and
  override FBETA_BETA / FINAL_THRESHOLD_OVERRIDE below if you want a different balance.

BUG 4 (verified, not actually a bug, but now asserted explicitly) - "only use a
  physician's data up to their first adoption quarter".
  Traced this line by line: the original cum_adopt_before logic in build_lag_panel()
  already does this correctly - a physician's row at the quarter they first adopt IS
  kept (positive example), and every quarter after that is dropped. A new function,
  assert_single_adoption_per_physician(), now hard-fails the run if this is ever
  violated, so the logic is verifiable at runtime rather than taken on faith.

DESIGN ASSUMPTIONS (unchanged from v1, still valid):
  A1. A physician is only a valid "non-adopted" training example for target quarter Q if
      brand_prescribed == 0 for every quarter strictly before Q. Once a physician
      adopts, all subsequent quarters are dropped from the panel.
  A2. lag1 = most recent quarter strictly before target quarter; lag2 = second most
      recent. Built via groupby(physician_id).shift() -> the target quarter's own
      values are NEVER used as features -> zero leakage.
  A3. Rows where lag1 is completely unavailable (physician's very first observed
      quarter) are dropped. Rows where only lag2 is unavailable are kept; lag_avg /
      lag_change are left NaN for those rows.
  A4. For scoring Test_physicians.csv (Q11 prediction), lag1 = Quarter 10, lag2 =
      Quarter 9 for every test physician.
  A5. All EDA decisions (missingness threshold, gender mode, VIF) are learned on TRAIN
      ONLY and simply applied to test / the Q11 scoring set.

Run order: organized as sequential STEP blocks. Edit FILE PATHS in STEP 0, run top to
bottom.
=========================================================================================
"""

# =========================================================================================
# STEP 2 : ORDINAL ENCODE PHYSICIAN_VALUE_TIER (before lagging)
# =========================================================================================
def encode_tier_ordinal(df):
    df = df.copy()
    tiers_present = sorted(df[TIER_COL].dropna().unique().tolist())

    def sort_key(label):
        try:
            return int(str(label).split("-")[0])
        except Exception:
            return str(label)

    tiers_sorted = sorted(tiers_present, key=sort_key)
    tier_map = {label: rank + 1 for rank, label in enumerate(tiers_sorted)}
    df[TIER_COL] = df[TIER_COL].map(tier_map)
    return df, tier_map


# =========================================================================================
# STEP 3 : BUILD THE LAG-FEATURE PANEL
# =========================================================================================
def build_lag_panel(file1, is_scoring=False):
    df = file1.sort_values([ID_COL, TIME_COL]).copy()
    grp = df.groupby(ID_COL, group_keys=False)

    all_metric_cols = LAG_AVG_CHANGE_COLS + LAG_CHANGE_ONLY_COLS + MOST_RECENT_ONLY_COLS + [TIER_COL]
    all_metric_cols = [c for c in all_metric_cols if c in df.columns]

    lag_frames = {ID_COL: df[ID_COL], TIME_COL: df[TIME_COL], TARGET_COL: df[TARGET_COL]}

    for col in all_metric_cols:
        lag1 = grp[col].shift(1)
        lag2 = grp[col].shift(2)
        lag_frames[f"{col}_lag1"] = lag1
        lag_frames[f"{col}_lag2"] = lag2

        if col in LAG_AVG_CHANGE_COLS:
            lag_frames[f"{col}_lag_avg"] = np.where(
                lag1.notna() & lag2.notna(), (lag1 + lag2) / 2, np.nan
            )
            lag_frames[f"{col}_lag_change"] = np.where(
                lag1.notna() & lag2.notna(), lag1 - lag2, np.nan
            )
        elif col in LAG_CHANGE_ONLY_COLS:
            lag_frames[f"{col}_lag_change"] = np.where(
                lag1.notna() & lag2.notna(), lag1 - lag2, np.nan
            )
        elif col == TIER_COL:
            lag_frames[f"{col}_lag_change"] = np.where(
                lag1.notna() & lag2.notna(), lag1 - lag2, np.nan
            )

    panel = pd.DataFrame(lag_frames)

    for col in MOST_RECENT_ONLY_COLS:
        panel[col] = panel[f"{col}_lag1"]
        panel.drop(columns=[f"{col}_lag1", f"{col}_lag2"], inplace=True, errors="ignore")

    if not is_scoring:
        # ---- A1: keep only rows where the physician had NOT adopted in any prior quarter ----
        cum_adopt_before = grp[TARGET_COL].apply(lambda s: s.shift(1).fillna(0).cumsum())
        panel["_adopted_before"] = (cum_adopt_before.values > 0)
        panel = panel[~panel["_adopted_before"]].drop(columns=["_adopted_before"])

        # ---- A3: drop rows with no history at all (lag1 fully missing) ----
        lag1_cols = [c for c in panel.columns if c.endswith("_lag1")]
        panel = panel.dropna(subset=lag1_cols, how="all")

        panel = panel.rename(columns={TIME_COL: "target_quarter"})
        panel = panel.reset_index(drop=True)
        assert_single_adoption_per_physician(panel)
        return panel

    else:
        panel = panel.rename(columns={TIME_COL: "target_quarter"})
        panel = panel.sort_values([ID_COL, "target_quarter"]).groupby(ID_COL).tail(1)
        panel = panel.drop(columns=[TARGET_COL])
        return panel.reset_index(drop=True)


def assert_single_adoption_per_physician(panel):
    """
    Verifies BUG 4 / A1 is actually enforced: no physician appears more than once with
    target==1, and (implicitly, since later rows were dropped) no physician has a row
    after their adoption quarter. Hard-fails loudly if violated so this can never
    silently regress.
    """
    positive_counts = panel[panel[TARGET_COL] == 1].groupby(ID_COL).size()
    violators = positive_counts[positive_counts > 1]
    if len(violators) > 0:
        raise AssertionError(
            f"[SANITY CHECK FAILED] {len(violators)} physician(s) have more than one "
            f"positive (adoption) row in the training panel - post-adoption quarters "
            f"were not correctly dropped. Example IDs: {violators.index[:5].tolist()}"
        )
    n_pos_phys = positive_counts.shape[0]
    print(f"[Sanity] Verified: every physician has at most 1 adoption row in the panel "
          f"({n_pos_phys} physicians adopt at some point in the observed history).")

# STEP 4 : MERGE WITH STATIC FILE2 + ONE-HOT ENCODE CATEGORICALS
# =========================================================================================
def merge_static_and_encode(panel, file2, ohe=None, fit_ohe=False):
    merged = panel.merge(file2, on=ID_COL, how="left")

    if fit_ohe:
        ohe = OneHotEncoder(handle_unknown="ignore", sparse_output=False)
        ohe.fit(merged[CATEGORICAL_COLS])

    encoded = ohe.transform(merged[CATEGORICAL_COLS])
    encoded_df = pd.DataFrame(
        encoded, columns=ohe.get_feature_names_out(CATEGORICAL_COLS), index=merged.index
    )
    merged = pd.concat([merged.drop(columns=CATEGORICAL_COLS), encoded_df], axis=1)
    return merged, ohe


# =========================================================================================
# STEP 5 : EDA - MISSING VALUE HANDLING, GENDER MODE IMPUTE, VIF, CORRELATION
# =========================================================================================
def eda_missing_value_drop(train_df, test_df, threshold=0.20):
    miss_pct = train_df.isna().mean()
    cols_to_drop = miss_pct[miss_pct > threshold].index.tolist()
    cols_to_drop = [c for c in cols_to_drop if c not in NON_FEATURE_COLS]
    print(f"[EDA] Dropping {len(cols_to_drop)} columns with >{threshold*100:.0f}% missing "
          f"(train-derived): {cols_to_drop}")
    train_df = train_df.drop(columns=cols_to_drop)
    test_df = test_df.drop(columns=[c for c in cols_to_drop if c in test_df.columns])
    return train_df, test_df, cols_to_drop


def impute_gender_mode(train_df, test_df, unknown_tokens=("Unknown", "unknown", "U", "?")):
    gender_cols = [c for c in train_df.columns if c.startswith("physician_gender_")]
    if not gender_cols:
        return train_df, test_df

    unknown_cols = [c for c in gender_cols if any(tok in c for tok in unknown_tokens)]
    if unknown_cols:
        mode_col = train_df[[c for c in gender_cols if c not in unknown_cols]].sum().idxmax()
        for udf, name in [(train_df, "train"), (test_df, "test")]:
            for ucol in unknown_cols:
                mask = udf[ucol] == 1
                udf.loc[mask, mode_col] = 1
                udf.loc[mask, ucol] = 0
    return train_df, test_df


def compute_vif(df, feature_cols):
    # NOTE: -999 / 0 fill here is ONLY for the VIF diagnostic matrix, never for modeling.
    X = df[feature_cols].fillna(0).astype(float)
    vif_data = pd.DataFrame()
    vif_data["feature"] = feature_cols
    vif_data["VIF"] = [variance_inflation_factor(X.values, i) for i in range(X.shape[1])]
    return vif_data.sort_values("VIF", ascending=False)


def vif_reduce(train_df, test_df, feature_cols, vif_threshold=10.0, max_iter=50):
    cols = feature_cols.copy()
    for _ in range(max_iter):
        vif_df = compute_vif(train_df, cols)
        worst = vif_df.iloc[0]
        if worst["VIF"] <= vif_threshold or len(cols) <= 1:
            break
        print(f"[VIF] Dropping '{worst['feature']}' (VIF={worst['VIF']:.1f})")
        cols.remove(worst["feature"])

    dropped = [c for c in feature_cols if c not in cols]
    train_df = train_df.drop(columns=dropped)
    test_df = test_df.drop(columns=[c for c in dropped if c in test_df.columns])
    return train_df, test_df, cols, dropped

# =========================================================================================
# STEP 6 : ROLLING-WINDOW (TIME-BASED) CROSS VALIDATION SPLITS
# =========================================================================================
def rolling_window_splits(df, quarter_col="target_quarter", min_train_quarters=3):
    quarters = sorted(df[quarter_col].unique())
    splits = []
    for i in range(min_train_quarters, len(quarters)):
        train_quarters = quarters[:i]
        val_quarter = quarters[i]
        train_idx = df.index[df[quarter_col].isin(train_quarters)]
        val_idx = df.index[df[quarter_col] == val_quarter]
        if len(val_idx) > 0 and df.loc[val_idx, TARGET_COL].nunique() > 1:
            splits.append((train_idx, val_idx))
    return splits


# =========================================================================================
# STEP 7 : OPTUNA HYPERPARAMETER TUNING (XGBoost & LightGBM)
#          FIX: no more SMOTE. Imbalance handled via scale_pos_weight, tuned as a
#          hyperparameter. Objective switched to PR-AUC (average precision), which is
#          the correct metric for rare-event ranking quality (ROC-AUC is easily
#          over-optimistic under heavy imbalance).
# =========================================================================================
def optuna_objective_factory(model_type, X, y, splits):
    def objective(trial):
        fold_scores = []
        for train_idx, val_idx in splits:
            X_tr, y_tr = X.loc[train_idx], y.loc[train_idx]
            X_val, y_val = X.loc[val_idx], y.loc[val_idx]

            neg, pos = (y_tr == 0).sum(), (y_tr == 1).sum()
            base_ratio = neg / max(pos, 1)
            spw_multiplier = trial.suggest_float("spw_multiplier", 0.5, 1.5)
            scale_pos_weight = base_ratio * spw_multiplier

            if model_type == "xgb":
                params = {
                    "n_estimators": trial.suggest_int("n_estimators", 100, 600),
                    "max_depth": trial.suggest_int("max_depth", 3, 10),
                    "learning_rate": trial.suggest_float("learning_rate", 0.01, 0.3, log=True),
                    "subsample": trial.suggest_float("subsample", 0.6, 1.0),
                    "colsample_bytree": trial.suggest_float("colsample_bytree", 0.6, 1.0),
                    "min_child_weight": trial.suggest_int("min_child_weight", 1, 10),
                    "reg_alpha": trial.suggest_float("reg_alpha", 1e-3, 10, log=True),
                    "reg_lambda": trial.suggest_float("reg_lambda", 1e-3, 10, log=True),
                    "scale_pos_weight": scale_pos_weight,
                    "random_state": RANDOM_STATE,
                    "eval_metric": "aucpr",
                    "use_label_encoder": False,
                }
                model = xgb.XGBClassifier(**params)
            else:
                params = {
                    "n_estimators": trial.suggest_int("n_estimators", 100, 600),
                    "num_leaves": trial.suggest_int("num_leaves", 15, 128),
                    "learning_rate": trial.suggest_float("learning_rate", 0.01, 0.3, log=True),
                    "subsample": trial.suggest_float("subsample", 0.6, 1.0),
                    "colsample_bytree": trial.suggest_float("colsample_bytree", 0.6, 1.0),
                    "min_child_samples": trial.suggest_int("min_child_samples", 5, 50),
                    "reg_alpha": trial.suggest_float("reg_alpha", 1e-3, 10, log=True),
                    "reg_lambda": trial.suggest_float("reg_lambda", 1e-3, 10, log=True),
                    "scale_pos_weight": scale_pos_weight,
                    "random_state": RANDOM_STATE,
                    "verbosity": -1,
                }
                model = lgb.LGBMClassifier(**params)

            # NOTE: fit directly on X_tr / y_tr with NaN untouched - both models handle
            # missing values natively. No fillna(-999), no SMOTE.
            model.fit(X_tr, y_tr)
            preds = model.predict_proba(X_val)[:, 1]
            fold_scores.append(average_precision_score(y_val, preds))  # PR-AUC

        return float(np.mean(fold_scores))
    return objective


def tune_model(model_type, X, y, splits, n_trials=40):
    study = optuna.create_study(direction="maximize", sampler=optuna.samplers.TPESampler(seed=RANDOM_STATE))
    study.optimize(optuna_objective_factory(model_type, X, y, splits), n_trials=n_trials, show_progress_bar=False)
    print(f"[Optuna:{model_type}] Best PR-AUC (CV) = {study.best_value:.4f}")
    print(f"[Optuna:{model_type}] Best params = {study.best_params}")
    return study.best_params


# =========================================================================================
# STEP 7b : OUT-OF-FOLD PREDICTIONS + THRESHOLD OPTIMIZATION
#          FIX: replaces the hard-coded 0.5 cutoff. Threshold is learned from genuinely
#          out-of-sample predictions (rolling CV folds), never from the test set, so
#          reported test metrics stay honest. Same threshold is then reused everywhere:
#          train eval, test eval, and the final Q11 scoring output.
# =========================================================================================
def get_oof_predictions(model_type, best_params, X, y, splits):
    oof_index, oof_proba, oof_y = [], [], []
    for train_idx, val_idx in splits:
        X_tr, y_tr = X.loc[train_idx], y.loc[train_idx]
        X_val, y_val = X.loc[val_idx], y.loc[val_idx]

        neg, pos = (y_tr == 0).sum(), (y_tr == 1).sum()
        scale_pos_weight = (neg / max(pos, 1)) * best_params.get("spw_multiplier", 1.0)
        params = {k: v for k, v in best_params.items() if k != "spw_multiplier"}

        if model_type == "xgb":
            model = xgb.XGBClassifier(**params, scale_pos_weight=scale_pos_weight,
                                       random_state=RANDOM_STATE, eval_metric="aucpr",
                                       use_label_encoder=False)
        else:
            model = lgb.LGBMClassifier(**params, scale_pos_weight=scale_pos_weight,
                                        random_state=RANDOM_STATE, verbosity=-1)

        model.fit(X_tr, y_tr)
        proba = model.predict_proba(X_val)[:, 1]
        oof_index.extend(val_idx.tolist())
        oof_proba.extend(proba.tolist())
        oof_y.extend(y_val.tolist())

    return pd.Series(oof_proba, index=oof_index), pd.Series(oof_y, index=oof_index)


def find_best_threshold_fbeta(y_true, proba, beta=1.5):
    precisions, recalls, thresholds = precision_recall_curve(y_true, proba)
    precisions, recalls = precisions[:-1], recalls[:-1]  # align lengths with thresholds
    with np.errstate(divide="ignore", invalid="ignore"):
        fbeta = (1 + beta**2) * (precisions * recalls) / ((beta**2 * precisions) + recalls)
    fbeta = np.nan_to_num(fbeta, nan=0.0)
    best_idx = np.argmax(fbeta)
    return float(thresholds[best_idx]), precisions[best_idx], recalls[best_idx], fbeta[best_idx]


def print_threshold_scan(y_true, proba, label=""):
    print(f"\n[Threshold scan{(' - ' + label) if label else ''}] "
          f"(precision / recall / F1 / predicted-positive-count at each cutoff)")
    print(f"{'thresh':>7} {'precision':>10} {'recall':>8} {'f1':>7} {'n_pred_pos':>11}")
    for t in np.arange(0.10, 0.95, 0.05):
        preds = (proba >= t).astype(int)
        p = precision_score(y_true, preds, zero_division=0)
        r = recall_score(y_true, preds, zero_division=0)
        f = f1_score(y_true, preds, zero_division=0)
        print(f"{t:7.2f} {p:10.4f} {r:8.4f} {f:7.4f} {preds.sum():11d}")


# =========================================================================================
# STEP 8 : FINAL MODEL TRAINING (best params, scale_pos_weight, NO SMOTE)
# =========================================================================================
def fit_final_model(model_type, best_params, X_train, y_train):
    neg, pos = (y_train == 0).sum(), (y_train == 1).sum()
    scale_pos_weight = (neg / max(pos, 1)) * best_params.get("spw_multiplier", 1.0)
    params = {k: v for k, v in best_params.items() if k != "spw_multiplier"}

    if model_type == "xgb":
        model = xgb.XGBClassifier(**params, scale_pos_weight=scale_pos_weight,
                                   random_state=RANDOM_STATE, eval_metric="aucpr",
                                   use_label_encoder=False)
    else:
        model = lgb.LGBMClassifier(**params, scale_pos_weight=scale_pos_weight,
                                    random_state=RANDOM_STATE, verbosity=-1)

    # NOTE: fit on X_train with NaN untouched - no fillna(-999), no SMOTE.
    model.fit(X_train, y_train)
    return model


# =========================================================================================
# STEP 9 : FULL EVALUATION SUITE (classification + business metrics)
#          FIX: threshold is now a required, tuned argument (no more default 0.5).
# =========================================================================================
def evaluate_model(model, X, y, dataset_name, model_name, threshold):
    proba = model.predict_proba(X)[:, 1]  # NaN passed through natively
    preds = (proba >= threshold).astype(int)

    tn, fp, fn, tp = confusion_matrix(y, preds).ravel()
    specificity = tn / (tn + fp) if (tn + fp) > 0 else np.nan

    acc = accuracy_score(y, preds)
    prec = precision_score(y, preds, zero_division=0)
    rec = recall_score(y, preds, zero_division=0)
    f1 = f1_score(y, preds, zero_division=0)
    fbeta = fbeta_score(y, preds, beta=FBETA_BETA, zero_division=0)
    roc_auc = roc_auc_score(y, proba)
    pr_auc = average_precision_score(y, proba)
    mae = mean_absolute_error(y, proba)

    nonzero_mask = y > 0
    mape = (np.abs(y[nonzero_mask] - proba[nonzero_mask]) / y[nonzero_mask]).mean() * 100 if nonzero_mask.sum() > 0 else np.nan

    df_rank = pd.DataFrame({"y": y.values, "proba": proba}).sort_values("proba", ascending=False)
    df_rank["decile"] = pd.qcut(df_rank["proba"].rank(method="first", ascending=False), 10, labels=False)
    overall_rate = df_rank["y"].mean()
    top_decile_rate = df_rank[df_rank["decile"] == 0]["y"].mean()
    top_decile_lift = top_decile_rate / overall_rate if overall_rate > 0 else np.nan
    top20pct_n = int(len(df_rank) * 0.2)
    top20pct_capture = df_rank.head(top20pct_n)["y"].sum() / df_rank["y"].sum() if df_rank["y"].sum() > 0 else np.nan

    print(f"\n===== {model_name} | {dataset_name}  (threshold={threshold:.3f}) =====")
    print(f"Accuracy            : {acc:.4f}")
    print(f"Precision           : {prec:.4f}")
    print(f"Recall (Sensitivity): {rec:.4f}")
    print(f"Specificity         : {specificity:.4f}")
    print(f"F1-Score            : {f1:.4f}")
    print(f"F{FBETA_BETA}-Score           : {fbeta:.4f}")
    print(f"ROC-AUC             : {roc_auc:.4f}")
    print(f"PR-AUC              : {pr_auc:.4f}")
    print(f"MAE (on probability): {mae:.4f}")
    print(f"MAPE (positives only): {mape:.2f}%")
    print(f"Top-decile lift     : {top_decile_lift:.2f}x  (top-decile adoption rate = {top_decile_rate:.2%} "
          f"vs overall {overall_rate:.2%})")
    print(f"Top-20% capture rate: {top20pct_capture:.2%} of all adopters found in top 20% ranked physicians")
    print(classification_report(y, preds, zero_division=0))

    fpr, tpr, _ = roc_curve(y, proba)
    plt.figure(figsize=(6, 5))
    plt.plot(fpr, tpr, label=f"AUC = {roc_auc:.3f}")
    plt.plot([0, 1], [0, 1], linestyle="--", color="grey")
    plt.xlabel("False Positive Rate"); plt.ylabel("True Positive Rate")
    plt.title(f"ROC Curve - {model_name} ({dataset_name})")
    plt.legend(); plt.tight_layout()
    plt.savefig(f"roc_curve_{model_name}_{dataset_name}.png", dpi=150)
    plt.close()

    precision_arr, recall_arr, _ = precision_recall_curve(y, proba)
    plt.figure(figsize=(6, 5))
    plt.plot(recall_arr, precision_arr, label=f"PR-AUC = {pr_auc:.3f}")
    plt.xlabel("Recall"); plt.ylabel("Precision")
    plt.title(f"Precision-Recall Curve - {model_name} ({dataset_name})")
    plt.legend(); plt.tight_layout()
    plt.savefig(f"pr_curve_{model_name}_{dataset_name}.png", dpi=150)
    plt.close()

    return {
        "accuracy": acc, "precision": prec, "recall": rec, "specificity": specificity,
        "f1": f1, f"f{FBETA_BETA}": fbeta, "roc_auc": roc_auc, "pr_auc": pr_auc,
        "mae": mae, "mape": mape, "top_decile_lift": top_decile_lift,
        "top20pct_capture": top20pct_capture,
    }

# STEP 10 : SHAP EXPLAINABILITY
#          FIX: NaN passed through untouched (matches how the model was actually trained;
#          -999 previously made SHAP explain a model behavior that didn't reflect training).
# =========================================================================================
def run_shap_analysis(model, X_train, model_name, max_display=20):
    explainer = shap.TreeExplainer(model)
    shap_values = explainer.shap_values(X_train)

    plt.figure()
    shap.summary_plot(shap_values, X_train, max_display=max_display, show=False)
    plt.title(f"SHAP Feature Importance - {model_name}")
    plt.tight_layout()
    plt.savefig(f"shap_summary_{model_name}.png", dpi=150)
    plt.close()
    print(f"[SHAP] Summary plot saved for {model_name}")

    mean_abs_shap = np.abs(shap_values).mean(axis=0)
    importance_df = pd.DataFrame({
        "feature": X_train.columns, "mean_abs_shap": mean_abs_shap
    }).sort_values("mean_abs_shap", ascending=False)
    return importance_df


# =========================================================================================
# STEP 11 : MAIN ORCHESTRATION
# =========================================================================================
def main():
    # ---- 1) Load ----
    file1, file2, test_phys = load_raw_data()

    # ---- 2) Ordinal encode tier BEFORE lagging ----
    file1, tier_map = encode_tier_ordinal(file1)
    print(f"[Tier] Ordinal mapping learned: {tier_map}")

    # ---- 3) Build training panel ----
    train_panel = build_lag_panel(file1, is_scoring=False)
    print(f"[Panel] Training panel shape: {train_panel.shape}, "
          f"positive rate: {train_panel[TARGET_COL].mean():.3%}")

    # ---- 3b) Build Q11 scoring panel for Test_physicians ----
    score_panel_full = build_lag_panel(file1, is_scoring=True)
    score_panel = score_panel_full[score_panel_full[ID_COL].isin(test_phys[ID_COL])].copy()
    print(f"[Panel] Q11 scoring panel shape: {score_panel.shape} "
          f"(expected ~{len(test_phys)} physicians)")

    # ---- 4) Merge static file2 + one-hot encode categoricals ----
    train_merged, ohe = merge_static_and_encode(train_panel, file2, fit_ohe=True)
    score_merged, _ = merge_static_and_encode(score_panel, file2, ohe=ohe, fit_ohe=False)

    # ---- 5) Train/test split (80/20, stratified) ----
    feature_cols_all = [c for c in train_merged.columns if c not in NON_FEATURE_COLS]
    X_all = train_merged[feature_cols_all]
    y_all = train_merged[TARGET_COL]

    X_train, X_test, y_train, y_test, meta_train, meta_test = train_test_split(
        X_all, y_all, train_merged[[ID_COL, "target_quarter"]],
        test_size=0.20, stratify=y_all, random_state=RANDOM_STATE
    )
    train_df = pd.concat([meta_train, X_train, y_train], axis=1)
    test_df = pd.concat([meta_test, X_test, y_test], axis=1)

    # ---- 6) EDA: missingness drop, gender mode impute, VIF, correlation ----
    train_df, test_df, dropped_missing_cols = eda_missing_value_drop(train_df, test_df, threshold=0.20)
    train_df, test_df = impute_gender_mode(train_df, test_df)

    numeric_feature_cols = [
        c for c in train_df.columns
        if c not in NON_FEATURE_COLS and pd.api.types.is_numeric_dtype(train_df[c])
    ]
    train_vif_ready = train_df.copy()
    train_vif_ready[numeric_feature_cols] = train_vif_ready[numeric_feature_cols].fillna(
        train_vif_ready[numeric_feature_cols].median()
    )

    train_df, test_df, kept_features, dropped_vif_cols = vif_reduce(
        train_df, test_df, numeric_feature_cols, vif_threshold=10.0
    )
    print(f"[VIF] Final feature count after VIF reduction: {len(kept_features)}")
    plot_correlation_heatmap(train_vif_ready, kept_features)

    for col in kept_features:
        if col not in score_merged.columns:
            score_merged[col] = np.nan
    score_final = score_merged[[ID_COL] + kept_features]

    X_train_final = train_df[kept_features]
    y_train_final = train_df[TARGET_COL]
    X_test_final = test_df[kept_features]
    y_test_final = test_df[TARGET_COL]

    # ---- 7) Rolling window CV splits (TRAIN portion only) ----
    splits = rolling_window_splits(train_df, quarter_col="target_quarter", min_train_quarters=3)
    print(f"[CV] Built {len(splits)} rolling-window (expanding) validation folds")

    # ---- 8) Optuna tuning: XGBoost & LightGBM (PR-AUC objective, scale_pos_weight, no SMOTE) ----
    xgb_best_params = tune_model("xgb", X_train_final, y_train_final, splits, n_trials=40)
    lgb_best_params = tune_model("lgb", X_train_final, y_train_final, splits, n_trials=40)

    # ---- 9) Fit final models on full train (no SMOTE, native NaN handling) ----
    xgb_model = fit_final_model("xgb", xgb_best_params, X_train_final, y_train_final)
    lgb_model = fit_final_model("lgb", lgb_best_params, X_train_final, y_train_final)

    # ---- 9b) Out-of-fold predictions -> optimal threshold per model ----
    xgb_oof_proba, xgb_oof_y = get_oof_predictions("xgb", xgb_best_params, X_train_final, y_train_final, splits)
    lgb_oof_proba, lgb_oof_y = get_oof_predictions("lgb", lgb_best_params, X_train_final, y_train_final, splits)

    xgb_threshold, xp, xr, xfb = find_best_threshold_fbeta(xgb_oof_y, xgb_oof_proba, beta=FBETA_BETA)
    lgb_threshold, lp, lr, lfb = find_best_threshold_fbeta(lgb_oof_y, lgb_oof_proba, beta=FBETA_BETA)
    print(f"\n[Threshold] XGBoost  OOF-optimal threshold = {xgb_threshold:.3f} "
          f"(OOF precision={xp:.3f}, recall={xr:.3f}, F{FBETA_BETA}={xfb:.3f})")
    print(f"[Threshold] LightGBM OOF-optimal threshold = {lgb_threshold:.3f} "
          f"(OOF precision={lp:.3f}, recall={lr:.3f}, F{FBETA_BETA}={lfb:.3f})")

    print_threshold_scan(xgb_oof_y, xgb_oof_proba, label="XGBoost OOF")
    print_threshold_scan(lgb_oof_y, lgb_oof_proba, label="LightGBM OOF")

    if FINAL_THRESHOLD_OVERRIDE is not None:
        xgb_threshold = lgb_threshold = FINAL_THRESHOLD_OVERRIDE
        print(f"\n[Threshold] Override active - using fixed threshold {FINAL_THRESHOLD_OVERRIDE} for both models")

    # ---- 10) Evaluate both models on TRAIN and TEST at their tuned thresholds ----
    results = {}
    results[("XGBoost", "train")] = evaluate_model(xgb_model, X_train_final, y_train_final, "train", "XGBoost", xgb_threshold)
    results[("XGBoost", "test")] = evaluate_model(xgb_model, X_test_final, y_test_final, "test", "XGBoost", xgb_threshold)
    results[("LightGBM", "train")] = evaluate_model(lgb_model, X_train_final, y_train_final, "train", "LightGBM", lgb_threshold)
    results[("LightGBM", "test")] = evaluate_model(lgb_model, X_test_final, y_test_final, "test", "LightGBM", lgb_threshold)

    results_df = pd.DataFrame(results).T
    print("\n===== MODEL COMPARISON SUMMARY =====")
    print(results_df)

    # ---- 11) SHAP explainability ----
    xgb_importance = run_shap_analysis(xgb_model, X_train_final, "XGBoost")
    lgb_importance = run_shap_analysis(lgb_model, X_train_final, "LightGBM")
    print("\nTop 15 features (XGBoost, SHAP):\n", xgb_importance.head(15))
    print("\nTop 15 features (LightGBM, SHAP):\n", lgb_importance.head(15))

    # ---- 12) Pick best model by TEST PR-AUC (better than ROC-AUC for rare-event ranking) ----
    best_name = "XGBoost" if results[("XGBoost", "test")]["pr_auc"] >= results[("LightGBM", "test")]["pr_auc"] else "LightGBM"
    best_model = xgb_model if best_name == "XGBoost" else lgb_model
    best_threshold = xgb_threshold if best_name == "XGBoost" else lgb_threshold
    print(f"\n[Final] Selected best model based on test PR-AUC: {best_name} "
          f"(threshold = {best_threshold:.3f})")

    X_score = score_final[kept_features]
    score_proba = best_model.predict_proba(X_score)[:, 1]  # NaN passed through natively
    score_class = (score_proba >= best_threshold).astype(int)

    output = pd.DataFrame({
        ID_COL: score_final[ID_COL],
        "adoption_probability_Q11": score_proba,
        "predicted_adoption_class_Q11": score_class,
    }).sort_values("adoption_probability_Q11", ascending=False).reset_index(drop=True)

    # secondary business-friendly flag: top-N by probability, independent of threshold
    output["top_N_flag"] = 0
    output.loc[output.index[:TOPN_BUSINESS_FLAG], "top_N_flag"] = 1

    output.to_csv(OUTPUT_SCORED_PATH, index=False)

    n_pred_pos = score_class.sum()
    pred_rate = n_pred_pos / len(score_class)
    print(f"\n[Output] Scored file written to: {OUTPUT_SCORED_PATH}")
    print(f"[Output] Predicted positive (threshold={best_threshold:.3f}): "
          f"{n_pred_pos} / {len(score_class)} physicians ({pred_rate:.1%})")
    print(f"[Output] Top-{TOPN_BUSINESS_FLAG} business flag also included as 'top_N_flag' column "
          f"for a fixed-size candidate list regardless of threshold.")
    print(output.head(10))

    return {
        "train_df": train_df, "test_df": test_df,
        "xgb_model": xgb_model, "lgb_model": lgb_model,
        "xgb_threshold": xgb_threshold, "lgb_threshold": lgb_threshold,
        "results_df": results_df,
        "xgb_importance": xgb_importance, "lgb_importance": lgb_importance,
        "scored_output": output,
    }


if __name__ == "__main__":
    main()
