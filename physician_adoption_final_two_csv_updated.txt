# ============================================================
# END-TO-END PHARMA PHYSICIAN NEXT-QUARTER BRAND ADOPTION MODEL
# ============================================================
# PURPOSE
# Predict whether a physician who has NOT previously prescribed the
# brand will prescribe the brand in the NEXT observed quarter.
#
# MODELS ONLY:
#   1. XGBoost
#   2. LightGBM
#
# DESIGN PRINCIPLES
#   - Business-driven pharma feature engineering
#   - Physician-quarter aggregation using metric-specific business logic
#   - No physician gender / sex used as a predictor
#   - No future information used in features
#   - No lag1/lag2/lag3 feature explosion
#   - Recent history is represented through business-relevant rolling trends
#   - Advanced EDA: missingness, duplicates, class balance, distributions,
#     correlation, mutual information, VIF and redundancy diagnostics
#   - Correlation/VIF/feature selection are fitted using TRAIN only
#   - Temporal train/validation/test split
#   - One-hot encoding only for selected static categorical variables
#   - SMOTE is applied ONLY inside CV training folds and final training data
#   - Optuna: only 5 trials x 3-fold StratifiedKFold for speed
#   - PR-AUC is the primary tuning metric because adoption is usually imbalanced
#   - Threshold is selected on VALIDATION only; TEST remains untouched
#   - Train/Validation/Test evaluation with accuracy, precision, recall,
#     sensitivity, specificity, F1, ROC-AUC, PR-AUC and Brier score
#   - Overfitting gap analysis
#   - SHAP explainability for the selected model
#
# IMPORTANT
# 1. Put the input file in the same folder as this notebook/script.
# 2. The pipeline uses input_data_file1.csv and input_data_file2.csv.
# 3. File 2 is the static physician profile and is merged on physician_id.
# 4. The target is NEXT-QUARTER brand prescription/adoption, not the current
#    quarter's brand_prescribed value.
# ============================================================

def find_column(possible_names, dataframe):
    lower_map = {str(c).strip().lower(): c for c in dataframe.columns}
    for name in possible_names:
        key = str(name).strip().lower()
        if key in lower_map:
            return lower_map[key]
    return None

def find_contains(dataframe, include_terms, exclude_terms=None):
    exclude_terms = exclude_terms or []
    for c in dataframe.columns:
        text = str(c).strip().lower()
        if any(term in text for term in include_terms) and not any(term in text for term in exclude_terms):
            return c
    return None

def choose_column(manual, candidates, dataframe, required=True):
    if manual is not None:
        if manual not in dataframe.columns:
            raise ValueError(f"Configured column '{manual}' not found.")
        return manual
    result = find_column(candidates, dataframe)
    if required and result is None:
        raise ValueError(
            "Could not detect a required column. Set it manually in CONFIG.\n"
            f"Candidates checked: {candidates}"
        )
    return result

def first_existing(candidates, dataframe):
    return find_column(candidates, dataframe)

def convert_binary(series):
    if pd.api.types.is_numeric_dtype(series):
        out = pd.to_numeric(series, errors="coerce")
        return np.where(out.isna(), np.nan, (out > 0).astype(float))
    mapping = {
        "yes": 1, "y": 1, "true": 1, "1": 1,
        "adopted": 1, "prescribed": 1, "brand prescribed": 1,
        "no": 0, "n": 0, "false": 0, "0": 0,
        "not adopted": 0, "not_prescribed": 0, "not prescribed": 0
    }
    return series.astype(str).str.strip().str.lower().map(mapping)

def make_time_index(series):
    # Numeric quarter index is already ideal.
    if pd.api.types.is_numeric_dtype(series):
        numeric = pd.to_numeric(series, errors="coerce")
        return numeric, numeric.astype(str)

    s = series.astype(str).str.strip()
    out = pd.Series(np.nan, index=series.index, dtype=float)
    labels = pd.Series(pd.NA, index=series.index, dtype="object")

    # Patterns such as 2021Q1, Q1 2021, Q1-2021.
    for idx, value in s.items():
        match = re.search(r"(20\d{2}).*?[Qq]([1-4])", value)
        if not match:
            match = re.search(r"[Qq]([1-4]).*?(20\d{2})", value)
            if match:
                q = int(match.group(1))
                year = int(match.group(2))
            else:
                q = None
                year = None
        else:
            year = int(match.group(1))
            q = int(match.group(2))
        if year is not None:
            out.loc[idx] = year * 4 + q
            labels.loc[idx] = f"{year}Q{q}"

    # Remaining values: parse as dates.
    remaining = out.isna()
    parsed = pd.to_datetime(s.where(remaining), errors="coerce")
    valid = parsed.notna()
    out.loc[valid] = parsed.loc[valid].dt.year * 4 + parsed.loc[valid].dt.quarter
    labels.loc[valid] = parsed.loc[valid].dt.to_period("Q").astype(str)

    # Final fallback: sorted string order.
    if out.isna().any():
        vals = sorted(s[out.isna()].dropna().unique())
        mapping = {v: i for i, v in enumerate(vals)}
        for value, idx in mapping.items():
            mask = out.isna() & s.eq(value)
            out.loc[mask] = 100000 + idx
            labels.loc[mask] = value

    return out, labels

# ============================================================
# 9. PHYSICIAN-QUARTER AGGREGATION WITH BUSINESS LOGIC
# ============================================================
# This is deliberately NOT a blind SUM of every numeric column.
#
# Business logic:
#   - Activity/count/volume variables -> SUM across events in a quarter.
#   - Binary brand-prescribed source -> MAX (any prescription/adoption event).
#   - Static physician descriptors -> first non-null / mode.
#   - Rate/percentage/index-like numeric fields -> MEAN, not SUM.
#   - IDs and time fields -> first.
#
# This prevents nonsensical aggregation of ratios, percentages and IDs.

count_measure_cols = [
    calls_col, new_rx_col, indication_col, competitor_col, samples_col,
    voucher_col, seminar_attendee_col, seminar_speaker_col, brand_paid_col,
    brand_organic_col, digital_visits_col, digital_impressions_col,
    commercial_col, medicare_col, medicaid_col
]
count_measure_cols = [c for c in count_measure_cols if c is not None]

# Detect likely rate / percentage variables so they are never summed.
rate_cols = []
for c in df.select_dtypes(include=np.number).columns:
    text = str(c).lower()
    if any(k in text for k in ["rate", "pct", "percent", "ratio", "share", "margin"]):
        rate_cols.append(c)

# Only apply metric-specific aggregation if duplicate physician-quarter rows exist.

        find_column(["physician_specialty", "specialty", "speciality", "hcp_specialty"], df),
        find_column(["region", "geography", "territory", "area"], df),
        find_column(["physician_gender", "gender", "sex"], df)
    ]
    static_candidates = [c for c in static_candidates if c is not None]

    agg = {}
    for c in count_measure_cols:
        if c in rate_cols:
            agg[c] = "mean"
        else:
            agg[c] = "sum"

    # Preserve other numeric measures as mean rather than blindly summing them.
    protected = set(count_measure_cols + [TARGET_SOURCE_COL, "_time_index"])
    for c in df.select_dtypes(include=np.number).columns:
        if c not in protected:
            agg[c] = "mean"

    agg[TARGET_SOURCE_COL] = "max"
    agg["_time_index"] = "first"
    agg["_time_label"] = "first"

    for c in static_candidates:
        agg[c] = lambda s: s.dropna().mode().iloc[0] if not s.dropna().empty else np.nan

    df = df.groupby([ID_COL, "_time_index"], as_index=False).agg(agg)
    df[QUARTER_COL] = df["_time_label"]

def safe_divide(a, b):
    b = b.replace(0, np.nan)
    out = a / b
    return out.replace([np.inf, -np.inf], np.nan)

    df[name] = pd.to_numeric(series, errors="coerce")

# ============================================================
# 12. BUSINESS-RELEVANT HISTORY (NO LAG FEATURE EXPLOSION)
# ============================================================
# We do NOT create lag1, lag2, lag3 columns for every variable.
# Instead, only business-relevant history is summarized.
# shift(1) is critical: the current quarter never uses future data.

def add_prior_rolling_mean(source_col, feature_name, window=3):
    if source_col is None or source_col not in df.columns:
        return
    prior = (
        df.groupby(ID_COL)[source_col]
        .shift(1)
        .groupby(df[ID_COL])
        .transform(lambda s: s.rolling(window, min_periods=1).mean())
    )
    add_feature(feature_name, prior)

if indication_col:

# 27. FAST OPTUNA SEARCH SPACES
# ============================================================
def objective_factory(model_name):
    def objective(trial):
        if model_name == "xgboost":
            params = {
                "n_estimators": trial.suggest_int("n_estimators", 80, 220),
                "max_depth": trial.suggest_int("max_depth", 2, 5),
                "learning_rate": trial.suggest_float("learning_rate", 0.03, 0.15, log=True),
                "subsample": trial.suggest_float("subsample", 0.75, 1.0),
                "colsample_bytree": trial.suggest_float("colsample_bytree", 0.70, 1.0),
                "min_child_weight": trial.suggest_int("min_child_weight", 2, 10),
                "reg_alpha": trial.suggest_float("reg_alpha", 1e-3, 2.0, log=True),
                "reg_lambda": trial.suggest_float("reg_lambda", 0.1, 10.0, log=True)
            }
        else:
            params = {
                "n_estimators": trial.suggest_int("n_estimators", 80, 220),
                "num_leaves": trial.suggest_int("num_leaves", 7, 31),
                "max_depth": trial.suggest_int("max_depth", 2, 7),
                "learning_rate": trial.suggest_float("learning_rate", 0.03, 0.15, log=True),
                "subsample": trial.suggest_float("subsample", 0.75, 1.0),
                "colsample_bytree": trial.suggest_float("colsample_bytree", 0.70, 1.0),
                "min_child_samples": trial.suggest_int("min_child_samples", 10, 50),
                "reg_alpha": trial.suggest_float("reg_alpha", 1e-3, 2.0, log=True),
                "reg_lambda": trial.suggest_float("reg_lambda", 0.1, 10.0, log=True)
            }

        fold_scores = []
        for train_idx, valid_idx in cv.split(X_train_processed, y_train):
            X_tr = X_train_processed[train_idx]
            y_tr = y_train.iloc[train_idx].to_numpy()
            X_cv = X_train_processed[valid_idx]
            y_cv = y_train.iloc[valid_idx].to_numpy()

            # SMOTE is applied only to the training fold.
            # Validation fold remains completely untouched.
            minority_count = int(pd.Series(y_tr).value_counts().min())
            if minority_count >= 2:
                k_neighbors = min(5, minority_count - 1)
                smote = SMOTE(random_state=RANDOM_STATE, k_neighbors=k_neighbors)
                X_tr_smote, y_tr_smote = smote.fit_resample(X_tr, y_tr)
            else:
                X_tr_smote, y_tr_smote = X_tr, y_tr

            model = build_model(model_name, params)
            model.fit(X_tr_smote, y_tr_smote)
            prob = model.predict_proba(X_cv)[:, 1]
            fold_scores.append(average_precision_score(y_cv, prob))

        return float(np.mean(fold_scores))
    return objective

# 30. METRIC FUNCTIONS

def calculate_metrics(y_true, probability, threshold=0.50):
    pred = (probability >= threshold).astype(int)
    tn, fp, fn, tp = confusion_matrix(y_true, pred, labels=[0, 1]).ravel()
    sensitivity = tp / (tp + fn) if (tp + fn) else 0.0
    specificity = tn / (tn + fp) if (tn + fp) else 0.0
    return {
        "Accuracy": accuracy_score(y_true, pred),
        "Precision": precision_score(y_true, pred, zero_division=0),
        "Recall": recall_score(y_true, pred, zero_division=0),
        "Sensitivity": sensitivity,
        "Specificity": specificity,
        "F1": f1_score(y_true, pred, zero_division=0),
        "ROC_AUC": roc_auc_score(y_true, probability),
        "PR_AUC": average_precision_score(y_true, probability),
        "Brier_Score": brier_score_loss(y_true, probability)
    }

def choose_validation_threshold(y_true, probability):
    # Threshold selection is performed ONLY on validation.
    # Optimize F1 as a balanced precision/recall decision rule.
    thresholds = np.linspace(0.10, 0.90, 81)
    scores = []
    for threshold in thresholds:
        pred = (probability >= threshold).astype(int)
        scores.append(f1_score(y_true, pred, zero_division=0))
    return float(thresholds[int(np.argmax(scores))])

for model_name, model in final_models.items():

    val_prob = model.predict_proba(X_val_processed)[:, 1]
    test_prob = model.predict_proba(X_test_processed)[:, 1]

    threshold = choose_validation_threshold(y_val, val_prob)
    thresholds[model_name] = threshold

    print("\n" + "=" * 90)
    print(f"{model_name.upper()} | validation-selected threshold = {threshold:.2f}")
    print("=" * 90)

    for dataset_name, y_true, prob in [
        ("Train", y_train, train_prob),
        ("Validation", y_val, val_prob),
        ("Test", y_test, test_prob)
    ]:
        row = calculate_metrics(y_true, prob, threshold)
        row["Model"] = model_name
        row["Dataset"] = dataset_name
        row["Threshold"] = threshold
        results.append(row)
        print(dataset_name, {k: round(v, 4) for k, v in row.items() if isinstance(v, float)})

    probability_store[model_name] = {
        "train": train_prob,
        "validation": val_prob,
        "test": test_prob
    }

results_df = pd.DataFrame(results)
results_df = results_df[
    ["Model", "Dataset", "Threshold", "Accuracy", "Precision", "Recall",
     "Sensitivity", "Specificity", "F1", "ROC_AUC", "PR_AUC", "Brier_Score"]
]

print("\nFINAL MODEL COMPARISON")
display(results_df)

# ============================================================
# 32. OVERFITTING DIAGNOSTICS
# ============================================================
overfit_rows = []
for model_name in model_names:
    tr = results_df[(results_df.Model == model_name) & (results_df.Dataset == "Train")].iloc[0]
    va = results_df[(results_df.Model == model_name) & (results_df.Dataset == "Validation")].iloc[0]
    te = results_df[(results_df.Model == model_name) & (results_df.Dataset == "Test")].iloc[0]

    roc_train_val_gap = tr.ROC_AUC - va.ROC_AUC
    roc_val_test_gap = va.ROC_AUC - te.ROC_AUC
    pr_train_val_gap = tr.PR_AUC - va.PR_AUC
    pr_val_test_gap = va.PR_AUC - te.PR_AUC

    if max(abs(roc_train_val_gap), abs(pr_train_val_gap)) < 0.05:
        level = "Low"
    elif max(abs(roc_train_val_gap), abs(pr_train_val_gap)) < 0.10:
        level = "Moderate"
    else:
        level = "High"

    overfit_rows.append({
        "Model": model_name,
        "Train_Validation_ROC_Gap": roc_train_val_gap,
        "Validation_Test_ROC_Gap": roc_val_test_gap,
        "Train_Validation_PR_Gap": pr_train_val_gap,
        "Validation_Test_PR_Gap": pr_val_test_gap,
        "Overfitting_Level": level
    })

overfit_df = pd.DataFrame(overfit_rows)
print("\nOVERFITTING DIAGNOSTICS")
display(overfit_df)
